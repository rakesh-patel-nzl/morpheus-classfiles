LAB 4.2  CONFIGURE MYSQL                                             (pp.59-62)
================================================================================
RUN ON: lab-db-1, lab-db-2, lab-db-3  (all three)

mysql -u root -p

# Step 1 [MYSQL] Basic hardening
DELETE FROM mysql.user WHERE User='';
DROP DATABASE IF EXISTS test;
DELETE FROM mysql.db WHERE Db='test' OR Db='test\\_%';

# Step 2 [MYSQL] Enable invisible primary keys, then confirm value = ON
set persist sql_generate_invisible_primary_key=1;
SHOW GLOBAL VARIABLES LIKE 'sql_generate_invisible_primary_key';

# Step 3 [MYSQL] Create the Cluster Admin account
CREATE USER 'clusterAdmin'@'%' IDENTIFIED BY 'Password123?';
GRANT ALL PRIVILEGES ON *.* TO 'clusterAdmin'@'%' with grant option;
FLUSH PRIVILEGES;
quit

# Step 4 - Open mysqld.cnf for editing
sudo vi /etc/mysql/mysql.conf.d/mysqld.cnf

# Steps 5 & 6 - FILE EDIT: /etc/mysql/mysql.conf.d/mysqld.cnf
#   Add these lines UNDER the existing [mysqld] section (after log-error),
#   then add the [mysqldump] section at the end of the file.
----8<---- (paste under [mysqld]) ----
bind-address = 0.0.0.0
max_connections = 451
innodb_buffer_pool_size=12G
innodb_buffer_pool_instances=12
innodb_use_fdatasync=ON
sql_generate_invisible_primary_key=1
binlog_expire_logs_seconds=604800
binlog_expire_logs_auto_purge=ON
max_allowed_packet = 256M
default_time_zone = "+00:00"
character-set-server=utf8mb4
collation-server=utf8mb4_general_ci
----8<---- (paste at end of file) ----
[mysqldump]
set-gtid-purged=OFF
----8<----

# The finished section should look like this (p.60):
#   [mysqld]
#   pid-file        = /var/run/mysqld/mysqld.pid
#   socket          = /var/run/mysqld/mysqld.sock
#   datadir         = /var/lib/mysql
#   log-error       = /var/log/mysql/error.log
#   bind-address = 0.0.0.0
#   ... (lines above) ...
#   collation-server=utf8mb4_general_ci
#   [mysqldump]
#   set-gtid-purged=OFF

# Step 7 - Restart MySQL
sudo systemctl restart mysql

# Step 8 - Log in and verify the variables
mysql -u root -p

# [MYSQL]
show variables like "max_connections";
show variables like "max_allowed_packet";
select @@global.time_zone;
show variables like 'char%';
show variables like 'collation%';
SELECT @@innodb_buffer_pool_size;
SELECT @@innodb_buffer_pool_instances;
SELECT @@innodb_use_fdatasync;
show variables like "binlog_expire_logs_seconds";
show variables like "binlog_expire_logs_auto_purge";
quit

#   Finished section (p.110):
----8<----
#http/443
frontend http_front_tls
    bind *:443 ssl crt /etc/ssl/lab-lb.pem
    mode tcp
    log global
    option tcplog
    default_backend http_back_tls

backend http_back_tls
    balance source
    mode tcp
    log global
    option httpchk GET /ping
    http-check expect string MORPHEUS\ PING
    server lab-app-1 172.30.10.161:443 ssl verify none check port 80
    server lab-app-2 172.30.10.162:443 ssl verify none check port 80
    server lab-app-3 172.30.10.163:443 ssl verify none check port 80
